1. Concepto en lenguaje natural

¿Qué es una transacción?

Una transacción es un conjunto de operaciones que deben ejecutarse como una sola unidad. En palabras sencillas: todo se guarda o nada se guarda.

Ejemplo: Si se realiza una transferencia bancaria, no sirve descontar dinero de una cuenta si no se suma en la otra.

A - Atomicidad: todo o nada.

C - Consistencia: la base de datos queda válida.

I - Aislamiento: una operación no interfiere con otra.

D - Durabilidad: lo confirmado permanece.

Comandos principales

BEGIN / START TRANSACTION COMMIT ROLLBACK SAVEPOINT
ComandoSignificado
BEGINInicia la transacción.
COMMITConfirma los cambios.
ROLLBACKDeshace los cambios.
SAVEPOINTCrea un punto de regreso.

2Paso 2 — Crear tablas

Click para abrir/cerrar

Una tabla guarda cuentas, otra productos, otra ventas y otra detalle de ventas.

SQL
CREATE TABLE cuentas (
    id_cuenta SERIAL PRIMARY KEY,
    cliente VARCHAR(100) NOT NULL,
    saldo NUMERIC(12,2) NOT NULL
);

CREATE TABLE productos (
    id_producto SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    stock INT NOT NULL,
    precio NUMERIC(12,2) NOT NULL
);

CREATE TABLE ventas (
    id_venta SERIAL PRIMARY KEY,
    cliente VARCHAR(100),
    total NUMERIC(12,2),
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE detalle_ventas (
    id_detalle SERIAL PRIMARY KEY,
    id_venta INT REFERENCES ventas(id_venta),
    id_producto INT REFERENCES productos(id_producto),
    cantidad INT,
    subtotal NUMERIC(12,2)
);

4Paso 4 — Transacción con COMMIT

Click para abrir/cerrar

Se registra una venta, se descuenta stock y se descuenta saldo. Si todo sale bien, COMMIT confirma.

SQL
BEGIN;

INSERT INTO ventas (cliente, total)
VALUES ('Carlos Pérez', 90000)
RETURNING id_venta;

-- Suponga que el id_venta generado fue 1
INSERT INTO detalle_ventas (id_venta, id_producto, cantidad, subtotal)
VALUES (1, 1, 2, 90000);

UPDATE productos
SET stock = stock - 2
WHERE id_producto = 1;

UPDATE cuentas
SET saldo = saldo - 90000
WHERE cliente = 'Carlos Pérez';

COMMIT;

SELECT * FROM ventas;
SELECT * FROM detalle_ventas;
SELECT * FROM productos;
SELECT * FROM cuentas;

5Paso 5 — Transacción con ROLLBACK

Click para abrir/cerrar

Ahora se simula un error: se inicia la operación, pero se decide revertir.

SQL
BEGIN;

INSERT INTO ventas (cliente, total)
VALUES ('María López', 240000)
RETURNING id_venta;

-- Suponga que el id_venta generado fue 2
INSERT INTO detalle_ventas (id_venta, id_producto, cantidad, subtotal)
VALUES (2, 2, 2, 240000);

UPDATE productos
SET stock = stock - 2
WHERE id_producto = 2;

-- Algo salió mal: el cliente no autorizó el pago.
ROLLBACK;

-- Verifique que no se guardó la venta ni el descuento.
SELECT * FROM ventas;
SELECT * FROM productos;

6Paso 6 — SAVEPOINT

Click para abrir/cerrar

SAVEPOINT permite regresar a un punto sin cancelar toda la transacción.

SQL
BEGIN;

UPDATE cuentas
SET saldo = saldo - 50000
WHERE cliente = 'Carlos Pérez';

SAVEPOINT punto_descuento;

UPDATE productos
SET stock = stock - 20
WHERE id_producto = 1;

-- El descuento de stock fue exagerado, volvemos al punto anterior.
ROLLBACK TO SAVEPOINT punto_descuento;

COMMIT;

SELECT * FROM cuentas;
SELECT * FROM productos;

2Paso 2 — Crear tablas

Click para abrir/cerrar

En MySQL se usa AUTO_INCREMENT. Se recomienda motor InnoDB para transacciones.

SQL
CREATE TABLE cuentas (
    id_cuenta INT AUTO_INCREMENT PRIMARY KEY,
    cliente VARCHAR(100) NOT NULL,
    saldo DECIMAL(12,2) NOT NULL
) ENGINE=InnoDB;

CREATE TABLE productos (
    id_producto INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    stock INT NOT NULL,
    precio DECIMAL(12,2) NOT NULL
) ENGINE=InnoDB;

CREATE TABLE ventas (
    id_venta INT AUTO_INCREMENT PRIMARY KEY,
    cliente VARCHAR(100),
    total DECIMAL(12,2),
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE detalle_ventas (
    id_detalle INT AUTO_INCREMENT PRIMARY KEY,
    id_venta INT,
    id_producto INT,
    cantidad INT,
    subtotal DECIMAL(12,2),
    FOREIGN KEY (id_venta) REFERENCES ventas(id_venta),
    FOREIGN KEY (id_producto) REFERENCES productos(id_producto)
) ENGINE=InnoDB;
